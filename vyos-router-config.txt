# ============================================================================
# VyOS 1.5.x (Circinus) Zone-Based Router Configuration
# High Performance 10GbE Router Setup
# ============================================================================
#
# Hardware Configuration:
# - eth0, eth1: Aquantia AQC113C NICs (10GbE capable)
# - eth2-eth5: Intel I226-V NICs (2.5GbE capable)
#
# Network Layout:
# - WAN: eth0 (DHCP) - 20:7c:14:f8:35:8e
# - LAN: eth1 192.168.11.0/24 - 20:7c:14:f8:35:8f
# - LAN2: eth2 192.168.110.0/24 - 20:7c:14:f8:33:78
# - WIFI_TRUNK: eth3 - 20:7c:14:f8:33:79
#   - VLAN 30: WiFi (192.168.30.0/24)
#   - VLAN 40: WiFi Guest (192.168.40.0/24)
# - LAN4: eth4 - 20:7c:14:f8:33:7a
# - LAN5: eth5 - 20:7c:14:f8:33:7b
#
# Camera Group: 192.168.11.30 - 192.168.11.40
#
# Zones:
# - wan: External network (no inbound access)
# - lan: Primary LAN
# - wifi: WiFi VLAN (isolated from LAN)
# - wifi_guest: Guest WiFi VLAN (isolated from LAN)
# ============================================================================

# ----------------------------------------------------------------------------
# ENTER CONFIGURATION MODE
# ----------------------------------------------------------------------------
configure

# ----------------------------------------------------------------------------
# SYSTEM CONFIGURATION
# ----------------------------------------------------------------------------

set system domain-name 'local'
set system time-zone 'UTC'

# Enable hardware offload for performance

# INTERFACE CONFIGURATION
# ----------------------------------------------------------------------------

set interfaces ethernet eth0 offload gro
set interfaces ethernet eth0 offload gso
